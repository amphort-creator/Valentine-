<body>

    <h1 id="question">Will you be my Valentine? ❤️</h1>
    
    <div class="container">
        <button id="yesBtn" onclick="celebrate()">Yes</button>
        <button id="noBtn">No</button>
    </div>

    <script>
        const noBtn = document.getElementById('noBtn');
        const yesBtn = document.getElementById('yesBtn');
        const question = document.getElementById('question');

        // This makes the "No" button dodge the cursor/finger
        const handleEscape = () => {
            const btnWidth = noBtn.offsetWidth;
            const btnHeight = noBtn.offsetHeight;

            // Ensures it stays within the screen boundaries
            const maxX = window.innerWidth - btnWidth;
            const maxY = window.innerHeight - btnHeight;

            const randomX = Math.floor(Math.random() * maxX);
            const randomY = Math.floor(Math.random() * maxY);

            noBtn.style.position = 'fixed';
            noBtn.style.left = `${randomX}px`;
            noBtn.style.top = `${randomY}px`;
        };

        function celebrate() {
            question.innerText = "YAY! I knew you'd say yes! 💖";
            noBtn.style.display = 'none';
            yesBtn.style.transform = "scale(1.5)";

            // Confetti Explosion!
            const duration = 5 * 1000;
            const animationEnd = Date.now() + duration;
            const defaults = { startVelocity: 30, spread: 360, ticks: 60, zIndex: 0 };

            function randomInRange(min, max) {
              return Math.random() * (max - min) + min;
            }

            const interval = setInterval(function() {
              const timeLeft = animationEnd - Date.now();

              if (timeLeft <= 0) {
                return clearInterval(interval);
              }

              const particleCount = 50 * (timeLeft / duration);
              
              // Pink and Red hearts/confetti
              confetti(Object.assign({}, defaults, { 
                particleCount, 
                origin: { x: randomInRange(0.1, 0.3), y: Math.random() - 0.2 },
                colors: ['#ff0000', '#ff69b4', '#ff1493']
              }));
              confetti(Object.assign({}, defaults, { 
                particleCount, 
                origin: { x: randomInRange(0.7, 0.9), y: Math.random() - 0.2 },
                colors: ['#ff0000', '#ff69b4', '#ff1493']
              }));
            }, 250);
        }

        // Event listeners for the runaway button
        noBtn.addEventListener('mouseover', handleEscape);
        noBtn.addEventListener('touchstart', (e) => {
            e.preventDefault(); 
            handleEscape();
        });
        noBtn.addEventListener('click', handleEscape);
    </script>
